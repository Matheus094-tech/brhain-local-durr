(function () {
  // ATENÇÃO: os caminhos '/api/DashboardColaboradoresService/...' seguem o

  // '/api/DashboardService/obterVisaoGeral' a partir do módulo
  // 'brhain.rh.main.service.dashboard'). O nosso módulo se chama
  // 'brhain.rh.main.service.dashboardColaboradores', então por analogia o
  // nome de serviço vira 'DashboardColaboradoresService' — mas essa
  // convenção de nome não está documentada nos arquivos de referência, só
  // inferida por padrão. Se o endpoint não resolver, ajuste aqui pro nome
  // real que o RESTService.addEndpoint registrar no seu ambiente.
  var URL_OBTER_DADOS = '/api/RESTDashboard/obterDadosColaboradores';
  var URL_OBTER_OPCOES = '/api/RESTDashboard/obterOpcoesFiltroColaboradores';

  var PALETTE = ['#3b82f6', '#14b8a6', '#6366f1', '#f59e0b', '#22c55e', '#ef4444', '#0ea5e9', '#64748b', '#a855f7', '#e11d48'];
  var charts = {};

  if (window.Chart && window.ChartDataLabels) {
    Chart.register(ChartDataLabels);
  }

  function fmt(n) {
    n = Number(n || 0);
    return n.toLocaleString('pt-BR');
  }

  function destroyChart(key) {
    if (charts[key]) {
      charts[key].destroy();
      charts[key] = null;
    }
  }

  function barOptions(horizontal) {
    return {
      indexAxis: horizontal ? 'y' : 'x',
      responsive: true,
      maintainAspectRatio: false,
      plugins: {
        legend: { display: false },
        datalabels: {
          color: '#fff',
          anchor: 'end',
          align: 'start',
          clamp: true,
          font: { weight: '600', size: 11 },
          formatter: function (v) { return fmt(v); }
        },
        tooltip: {
          callbacks: {
            label: function (ctx) { return ' ' + fmt(ctx.raw); }
          }
        }
      },
      scales: {
        x: {
          grid: { color: '#eef2f7', drawBorder: false },
          ticks: { color: '#64748b', font: { size: 11 } },
          beginAtZero: true
        },
        y: {
          grid: { display: horizontal, color: '#eef2f7', drawBorder: false },
          ticks: { color: '#64748b', font: { size: 11 } },
          beginAtZero: true
        }
      }
    };
  }

  function makeBar(canvasId, key, labels, values, color, horizontal) {
    destroyChart(key);
    var colors = Array.isArray(color) ? color : labels.map(function () { return color; });
    charts[key] = new Chart(document.getElementById(canvasId).getContext('2d'), {
      type: 'bar',
      data: {
        labels: labels,
        datasets: [{
          data: values,
          backgroundColor: colors,
          borderRadius: 8,
          maxBarThickness: horizontal ? 22 : 44
        }]
      },
      options: barOptions(!!horizontal)
    });
  }

  function makeDonut(canvasId, key, labels, values, legendElId) {
    var sum = values.reduce(function (a, b) { return a + Number(b || 0); }, 0);
    destroyChart(key);
    charts[key] = new Chart(document.getElementById(canvasId).getContext('2d'), {
      type: 'doughnut',
      data: {
        labels: labels,
        datasets: [{
          data: values,
          backgroundColor: labels.map(function (_, i) { return PALETTE[i % PALETTE.length]; }),
          borderWidth: 0,
          hoverOffset: 4
        }]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        cutout: '68%',
        plugins: {
          legend: { display: false },
          datalabels: { display: false },
          tooltip: {
            callbacks: {
              label: function (ctx) {
                var v = Number(ctx.raw || 0);
                var pct = sum ? ((v / sum) * 100).toFixed(1) : '0.0';
                return ' ' + ctx.label + ': ' + pct + '% / ' + fmt(v);
              }
            }
          }
        }
      },
      plugins: [{
        id: 'centerText_' + key,
        afterDraw: function (chart) {
          var meta = chart.getDatasetMeta(0);
          if (!meta || !meta.data || !meta.data.length) return;
          var cx = meta.data[0].x;
          var cy = meta.data[0].y;
          var ctx = chart.ctx;
          ctx.save();
          ctx.textAlign = 'center';
          ctx.textBaseline = 'middle';
          ctx.fillStyle = '#1a2332';
          ctx.font = '700 22px Segoe UI, sans-serif';
          ctx.fillText(fmt(sum), cx, cy - 8);
          ctx.fillStyle = '#6b778c';
          ctx.font = '600 11px Segoe UI, sans-serif';
          ctx.fillText('total', cx, cy + 12);
          ctx.restore();
        }
      }]
    });
    if (legendElId) {
      var html = '';
      labels.forEach(function (label, i) {
        var v = values[i] || 0;
        var pct = sum ? ((v / sum) * 100).toFixed(1) : '0.0';
        html += '<li><span class="swatch" style="background:' + PALETTE[i % PALETTE.length] + '"></span>' +
          '<span>' + label + '</span><span class="pct">' + pct + '% / ' + fmt(v) + '</span></li>';
      });
      $('#' + legendElId).html(html || '<li>Sem dados</li>');
    }
  }

  function makeEvolucao(evolucao) {

          }
        ]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        interaction: { mode: 'index', intersect: false },
        plugins: {
          legend: {
            display: true,
            position: 'top',
            align: 'end',

          tooltip: {
            callbacks: {
              label: function (ctx) { return ' ' + ctx.dataset.label + ': ' + fmt(ctx.raw); }
            }
          }
        },
        scales: {
          x: {
            grid: { display: false },
            ticks: { color: '#64748b', font: { size: 11 } }

      }
    });
  }

  function renderKpis(k) {
    k = k || {};
    $('#kpiAtivos').text(fmt(k.colaboradoresAtivos));

  function renderAll(data) {
    data = data || {};
    $('#dashUpdated').text('Atualizado em ' + (data.atualizadoEm || '—'));
